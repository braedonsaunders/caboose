cmake_minimum_required(VERSION 3.16)
project(caboose_coherent_publication LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(caboose_coherent_publication STATIC publication.cpp)
target_compile_features(caboose_coherent_publication PUBLIC cxx_std_17)
target_include_directories(caboose_coherent_publication PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(caboose_coherent_publication PUBLIC Threads::Threads)
target_compile_options(caboose_coherent_publication PRIVATE -Wall -Wextra -Werror)
add_executable(publication_test publication_test.cpp)
target_link_libraries(publication_test PRIVATE caboose_coherent_publication)
enable_testing()
add_test(NAME publication COMMAND publication_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(caboose_protected_mapping STATIC protected_mapping.cpp)
    target_compile_features(caboose_protected_mapping PUBLIC cxx_std_17)
    target_link_libraries(caboose_protected_mapping PUBLIC Threads::Threads)
    target_compile_options(caboose_protected_mapping PRIVATE -Wall -Wextra -Werror)
    add_executable(protected_mapping_test protected_mapping_test.cpp)
    target_link_libraries(protected_mapping_test PRIVATE caboose_protected_mapping)
    add_test(NAME protected_mapping COMMAND protected_mapping_test)
    set_tests_properties(protected_mapping PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
endif()
