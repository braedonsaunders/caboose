cmake_minimum_required(VERSION 3.16)
project(caboose_coherent_publication LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(caboose_coherent_publication STATIC publication.cpp)
target_compile_features(caboose_coherent_publication PUBLIC cxx_std_17)
target_include_directories(caboose_coherent_publication PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(caboose_coherent_publication PUBLIC Threads::Threads)
target_compile_options(caboose_coherent_publication PRIVATE -Wall -Wextra -Werror)
add_executable(publication_test publication_test.cpp)
target_link_libraries(publication_test PRIVATE caboose_coherent_publication)
enable_testing()
add_test(NAME publication COMMAND publication_test)
